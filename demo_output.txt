=== Role: customer ===

   result={'ok': True, 'data': [{'id': 1, 'name': 'Dell XPS 13 Laptop', 'category': 'laptop', 'price': 999.99, 'stock': 5}, {'id': 2, 'name': 'MacBook Air M2', 'category': 'laptop', 'price': 1199.0, 'stock': 0}]}
